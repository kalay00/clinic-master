## 4.2 ECharts 图表组件

### 4.2.1 BaseChart 基础图表

```vue
<template>
  <BaseChart
    :option="chartOption"
    width="100%"
    height="300px"
    :loading="false"
    @chart-ready="onChartReady"
    @click="onChartClick"
  />
</template>

<script setup>
import { ref } from 'vue'
import BaseChart from '@/echarts/BaseChart.vue'

const chartOption = ref({
  xAxis: { type: 'category', data: ['Mon', 'Tue', 'Wed'] },
  yAxis: { type: 'value' },
  series: [{ data: [120, 200, 150], type: 'line' }]
})

function onChartReady(chart) {
  console.log('图表实例:', chart)
}

function onChartClick(params) {
  console.log('点击数据:', params)
}
</script>
```

### 4.2.2 useChart Composable

```vue
<template>
  <div ref="chartRef" style="width: 100%; height: 400px;"></div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { useChart } from '@/echarts/useChart'
import { getLineChartOption } from '@/echarts/chartOptions'

const chartRef = ref(null)

const { setOption, resize, destroyChart } = useChart(chartRef, {
  onReady: (chart) => {
    console.log('图表已就绪')
  }
})

onMounted(() => {
  const option = getLineChartOption({
    xData: ['周一', '周二', '周三', '周四', '周五'],
    series: [{ name: '销量', data: [120, 200, 150, 80, 70] }]
  })
  setOption(option)
})
</script>
```

### 4.2.3 图表配置工厂

```javascript
import {
  getLineChartOption,    // 折线图

// 柱状图
const barOption = getBarChartOption({

```

## 4.3 通用工具组件

### 4.3.1 消息提示

```javascript
import { toast, toastSuccess, toastError, confirm, alert } from '@/utils/message'

// 基础提示
toast('这是一条消息')

// 确认对话框

  try {
    await confirm('确定要删除吗？', '提示')
    // 用户点击确定
    console.log('确认删除')
  } catch {
    // 用户点击取消
  }
}

// 警告对话框
await alert('这是一个警告', '警告', 'warning')
```

### 4.3.2 本地存储

### 4.3.3 全局 Loading 遮罩

### 4.3.5 系统更新 (Electron)

提供 Electron 环境下的自动更新功能。

```javascript
import {
  checkForUpdates,
  downloadUpdate,
  installUpdate,
  isElectronEnv
} from '@/utils/update'

// 检测更新
const hasUpdate = await checkForUpdates()
if (hasUpdate) {
  console.log('发现新版本，正在下载...')
  await downloadUpdate()
}

// 安装更新并重启

```

浏览器环境下这些函数会返回安全的默认值。

### 4.3.6 快捷键支持 (useShortcut)

提供全局快捷键注册功能。

```javascript
import { useShortcut } from '@/composables/useShortcut'

const { register, unregister, unregisterAll } = useShortcut()

// 注册快捷键
register('ctrl+k', () => {
  console.log('Ctrl+K 被按下')
  // 打开搜索等操作
})

// 组合键
register('ctrl+s', (e) => {
  e.preventDefault()
  console.log('Ctrl+S 被按下')
  // 保存操作
})

// 卸载单个快捷键
unregister('ctrl+k')

// 卸载所有快捷键
unregisterAll()
```

## 4.4 自定义指令

### 4.4.1 v-permission 按钮级权限指令

用于控制按钮级别的权限显示。

```vue
<template>
  <!-- 单个权限 -->
  <el-button v-permission="['admin']" type="primary">
    管理按钮
  </el-button>

  <!-- 多个权限（满足任一即可） -->
  <el-button v-permission="['admin', 'editor']" type="success">
    编辑按钮
  </el-button>

  <!-- 带条件判断 -->
  <el-button v-permission="['admin']" v-if="hasAccess" type="danger">
    删除
  </el-button>
</template>

<script setup>
// 无需手动引入，指令已全局注册
</script>
```

指令参数可以是字符串数组或单个字符串：

```vue
<!-- 单个权限 -->
<el-button v-permission="'admin'">仅管理员</el-button>

<!-- 多个权限 -->
<el-button v-permission="['admin', 'editor']">管理员或编辑</el-button>
```

## 4.6 自定义组件创建

### 4.6.1 创建组件

在 `src/components/` 目录下创建组件：

```vue
<!-- src/components/MyComponent.vue -->
<template>
  <div class="my-component">
    <slot />
  </div>
</template>

<script setup>
defineProps({
  title: {
    type: String,
    default: ''
  }
})

defineEmits(['update'])
</script>

<style lang="scss" scoped>
.my-component {
  padding: 16px;
  background: #fff;
}
</style>
```

### 4.5.2 使用组件

```vue
<template>

    <p>插槽内容</p>
  </MyComponent>
</template>

<script setup>
function handleUpdate() {
  console.log('更新')
}
</script>
```

## 4.8 表单组件封装

```vue
<template>
  <el-form ref="formRef" :model="form" :rules="rules" label-width="100px">
    <el-form-item label="用户名" prop="username">
      <el-input v-model="form.username" placeholder="请输入用户名" />
    </el-form-item>
    <el-form-item label="邮箱" prop="email">
      <el-input v-model="form.email" placeholder="请输入邮箱" />
    </el-form-item>
    <el-form-item label="状态">
      <el-select v-model="form.status">
        <el-option label="启用" :value="1" />
        <el-option label="禁用" :value="0" />
      </el-select>
    </el-form-item>
    <el-form-item>
      <el-button type="primary" @click="handleSubmit">提交</el-button>
      <el-button @click="handleReset">重置</el-button>
    </el-form-item>
  </el-form>
</template>

<script setup>
import { ref, reactive } from 'vue'
import { validateRules } from '@/utils/validate'

const formRef = ref(null)
const form = reactive({
  username: '',
  email: '',
  status: 1
})

const rules = {
  username: [validateRules.required()],
  email: [validateRules.required(), validateRules.email()]
}

async function handleSubmit() {
  await formRef.value.validate()
  console.log('提交数据:', form)
}

function handleReset() {
  formRef.value.resetFields()
}
</script>
```
